-- ═══════════════════════════════════════════════
-- Agregar columnas a "Formulario de Candidatos"
-- ═══════════════════════════════════════════════
ALTER TABLE "Formulario de Candidatos"
  ADD COLUMN IF NOT EXISTS nombre                     TEXT,
  ADD COLUMN IF NOT EXISTS edad                       INTEGER,
  ADD COLUMN IF NOT EXISTS estatura                   NUMERIC,
  ADD COLUMN IF NOT EXISTS peso                       NUMERIC,
  ADD COLUMN IF NOT EXISTS genero                     TEXT,
  ADD COLUMN IF NOT EXISTS puesto_aplicado            TEXT,
  ADD COLUMN IF NOT EXISTS documentos                 JSONB DEFAULT '{}',
  ADD COLUMN IF NOT EXISTS requisitos_salud           JSONB DEFAULT '[]',
  ADD COLUMN IF NOT EXISTS respuestas_psicometricas   JSONB DEFAULT '{}',
  ADD COLUMN IF NOT EXISTS respuestas_tecnicas        JSONB DEFAULT '{}',
  ADD COLUMN IF NOT EXISTS puntaje_evaluacion_tecnica INTEGER,
  ADD COLUMN IF NOT EXISTS veredicto_final            TEXT,
  ADD COLUMN IF NOT EXISTS "timestamp"                TIMESTAMPTZ;

-- Habilitar acceso público
ALTER TABLE "Formulario de Candidatos" ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "allow_all" ON "Formulario de Candidatos";
CREATE POLICY "allow_all" ON "Formulario de Candidatos"
  FOR ALL USING (true) WITH CHECK (true);

-- Recargar schema cache
NOTIFY pgrst, 'reload schema';

-- Verificar resultado
SELECT column_name, data_type
FROM information_schema.columns
WHERE table_name = 'Formulario de Candidatos'
ORDER BY ordinal_position;
